export type TagColor = "red" | "green" | "orange" | "gray" | "blue" | "violet";

export interface StatusOption {
  value: string | number | boolean;
  label?: string; // text inside the tag (optional; defaults to value)
  color: TagColor; // one of 'red' | 'green' | 'orange' | 'gray'
}

export interface TableColumn {
  key: string; // unique field key
  label?: string; // header text (optional for backward-compat)
  header?: string; // backward-compat alias used by existing pages
  type?:
    | "string"
    | "date"
    | "time"
    | "boolean"
    | "custom"
    | "growth"
    | "actions"
    | "status"
    | "image";
  customTemplate?: any;
  searchable?: boolean; // show search input below header
  filterable?: boolean; // show filter icon button
  // Status tag configuration (used when type === 'status')
  statusOptions?: StatusOption[]; // mapping from raw values to tag label + color
  statusDefault?: { label?: string; color: TagColor }; // fallback when value not in options
}

export interface TableMetadata {
  pageNumber: number;
  totalCount: number;
  pageSize: number;
  totalPages: number;
}

export interface ActionOption {
  buttonName: string;
  icon: string;
}

export class TableComponent implements OnChanges {
  @Input() columns: TableColumn[] = [];
  // Allow undefined/null as an explicit "loading" state.
  @Input() data: any[] | undefined | null = undefined;
  @Input() metadata: TableMetadata | null = null;
  @Input() actionOptions: ActionOption[] = [];
  @Input() excludeActions: string[] = [];
  @Input() showSearchRow = true;
  @Input() searchDebounce = 300;
  @Input() searching = false; // dims tbody and shows preloader overlay when true
  @Input() useNzPagination = false; // use ng-zorro pagination instead of built-in
  @Input() imagePrefix: string = ""; // prefix for image paths when type === 'image'
  @Input() pageSizeOptions: number[] = [10, 20, 50, 100];
  @Input() filters: ReportTableFilterConfig[] = [];
  @Input() tableLabel = "";
  @Input() summaryLabel = "";
  @Input() summaryValue: string | number | null = null;
  @Input() showExportButton = false;
  @Input() showDownloadButton = false;

  @Output() pageChange = new EventEmitter<number>();
  @Output() pageSizeChange = new EventEmitter<number>();
  @Output() actionClick = new EventEmitter<{ action: number; item: any }>();
  @Output() searchChange = new EventEmitter<Record<string, string>>();
  @Output() filterClick = new EventEmitter<string>();
  @Output() filterChange = new EventEmitter<Record<string, any>>();
  @Output() exportClick = new EventEmitter<void>();
  @Output() downloadClick = new EventEmitter<void>();

  pages: number[] = [];
  columnQueries: Record<string, string> = {};
  sortKey: string | null = null;
  sortDir: "asc" | "desc" | null = null;
  filterSelections: Record<
    string,
    ReportTableFilterValue | ReportTableFilterValue[] | null
  > = {};
  // debounce timers per column
  private timers = new Map<string, ReturnType<typeof setTimeout>>();

  ngOnInit(): void {}
  ngOnChanges(changes: SimpleChanges): void {
    if (changes["metadata"] && this.metadata) {
      // Guard: totalPages should be at least 1
      const totalPages =
        this.metadata.totalPages && this.metadata.totalPages > 0
          ? this.metadata.totalPages
          : 1;
      this.pages = Array.from({ length: totalPages }, (_, i) => i + 1);
    }
    if (changes["filters"] && Array.isArray(this.filters)) {
      this.filters.forEach((filter) => {
        const hasSelection = Object.prototype.hasOwnProperty.call(
          this.filterSelections,
          filter.key,
        );
        if (!hasSelection) {
          if (typeof filter.value !== "undefined") {
            this.filterSelections[filter.key] = filter.value;
          } else if (
            filter.selection === "multi" &&
            Array.isArray(filter.values)
          ) {
            this.filterSelections[filter.key] = filter.values;
          }
        }
      });
    }
    // Note: data can be undefined/null to signal loading.
  }

  onFilterSelectionChange(key: string, value: any): void {
    this.filterSelections[key] = value;
    this.filterChange.emit({ ...this.filterSelections });
  }

  onExportClick(): void {
    this.exportClick.emit();
  }

  onDownloadClick(): void {
    this.downloadClick.emit();
  }

  isFilterAll(key: string): boolean {
    const value = this.filterSelections?.[key];
    return value === null || typeof value === "undefined";
  }

  isFilterActive(key: string, value: ReportTableFilterValue): boolean {
    const current = this.filterSelections?.[key];
    if (Array.isArray(current)) return current.includes(value);
    return current === value;
  }

  getFilterButtonLabel(filter: ReportTableFilterConfig): string {
    const value = this.filterSelections?.[filter.key];
    if (value === null || typeof value === "undefined") {
      return `${filter.label} (All)`;
    }
    if (Array.isArray(value)) {
      return `${filter.label} (${value.length})`;
    }
    const option = (filter.options ?? []).find((o) => o.value === value);
    return option?.label ?? filter.label;
  }

  onPageChange(page: number): void {
    if (
      this.metadata &&
      this.metadata.totalPages &&
      this.metadata.totalPages > 0 &&
      page >= 1 &&
      page <= this.metadata.totalPages &&
      page !== this.metadata.pageNumber
    ) {
      this.pageChange.emit(page);
    }
  }

  onPageSizeChange(size: number): void {
    this.pageSizeChange.emit(size);
  }

  onActionClick(action: number, item: any): void {
    this.actionClick.emit({ action, item });
  }

  onSearchInput(columnKey: string, value: string): void {
    this.columnQueries[columnKey] = value;

    // debounce per column
    const prev = this.timers.get(columnKey);
    if (prev) clearTimeout(prev);

    const t = setTimeout(() => {
      this.searchChange.emit({ ...this.columnQueries });
    }, this.searchDebounce);

    this.timers.set(columnKey, t);
  }

  onFilterClick(columnKey: string): void {
    this.filterClick.emit(columnKey);
  }

  onSort(column: TableColumn): void {
    if (!this.isSortable(column)) return;
    if (this.sortKey !== column.key) {
      this.sortKey = column.key;
      this.sortDir = "asc";
      return;
    }
    if (this.sortDir === "asc") {
      this.sortDir = "desc";
      return;
    }
    if (this.sortDir === "desc") {
      this.sortDir = null;
      this.sortKey = null;
      return;
    }
    this.sortDir = "asc";
  }

  isSortable(column: TableColumn): boolean {
    return column.type !== "actions" && column.type !== "custom";
  }

  getSortIcon(column: TableColumn): string {
    if (this.sortKey !== column.key || !this.sortDir)
      return "ri-arrow-up-down-line";
    return this.sortDir === "asc"
      ? "ri-arrow-up-s-line"
      : "ri-arrow-down-s-line";
  }

  get sortedData(): any[] {
    const data = this.data ?? [];
    if (!this.sortKey || !this.sortDir) return data;
    const column = this.columns.find((c) => c.key === this.sortKey);
    const dir = this.sortDir === "asc" ? 1 : -1;
    const sorted = [...data];
    sorted.sort((a, b) => {
      const av = this.getSortValue(a, this.sortKey!, column);
      const bv = this.getSortValue(b, this.sortKey!, column);
      if (av === null && bv === null) return 0;
      if (av === null) return 1 * dir;
      if (bv === null) return -1 * dir;
      if (av < bv) return -1 * dir;
      if (av > bv) return 1 * dir;
      return 0;
    });
    return sorted;
  }

  private getSortValue(item: any, key: string, column?: TableColumn): any {
    const raw = key.split(".").reduce((acc, part) => acc && acc[part], item);
    if (raw === undefined || raw === null) return null;
    if (column?.type === "date" || column?.type === "time") {
      const d = new Date(raw);
      return isNaN(d.getTime()) ? String(raw).toLowerCase() : d.getTime();
    }
    if (typeof raw === "number") return raw;
    return String(raw).trim().toLowerCase();
  }

  getCellValue(item: any, key: string, column?: TableColumn) {
    if (key === "No." && this.metadata) {
      const pageSize =
        this.metadata.pageSize && this.metadata.pageSize > 0
          ? this.metadata.pageSize
          : 1;
      const index = (this.data ?? []).indexOf(item);
      return (
        (this.metadata.pageNumber - 1) * pageSize + (index >= 0 ? index : 0) + 1
      );
    }
    const value = key.split(".").reduce((acc, part) => acc && acc[part], item);
    if (column?.type === "time" && value) {
      // Format ISO datetime as h:mm A
      const date = new Date(value);
      if (!isNaN(date.getTime())) {
        let hours = date.getHours();
        const minutes = date.getMinutes();
        const ampm = hours >= 12 ? "PM" : "AM";
        hours = hours % 12;
        hours = hours ? hours : 12; // the hour '0' should be '12'
        const mins = minutes < 10 ? "0" + minutes : minutes;
        return `${hours}:${mins} ${ampm}`;
      }
    }
    return value;
  }

  private toKey(val: any): string {
    return String(val).trim().toLowerCase();
  }

  getStatusTag(
    column: TableColumn,
    value: any,
  ): { label: string; color: TagColor } {
    const opts = column.statusOptions || [];
    const key = this.toKey(value);
    const match = opts.find((o) => this.toKey(o.value) === key);
    if (match) {
      return {
        label: match.label ?? String(value),
        color: match.color,
      };
    }
    const fallbackColor = column.statusDefault?.color ?? "gray";
    const fallbackLabel = column.statusDefault?.label ?? String(value);
    return { label: fallbackLabel, color: fallbackColor };
  }

  buildImageSrc(raw: any): string {
    return prefixImageUrl(raw);
  }

  onImgError(event: Event): void {
    const target = event.target as HTMLImageElement;
    target.src = prefixImageUrl(null);
  }
}
